# Looks at the app and writes down what is on the screen at every step.
#
# The names are written to a file, so that the steps of the real tour can be
# written with the names the app really shows.
#
# The taps are done by name and not by point, so that the look is not thrown
# off by a screen of another size.

shot 01-list
dumpfile 01-list
wait 2

# The + button in the corner.
taptext "New file" 02-new-file
wait 3
shot 03-editor
dumpfile 03-editor
wait 2

# The arrow back in the top bar of the editor.
taptext "Back" 04-back-from-the-editor
wait 3
shot 05-list-again
dumpfile 05-list-again
wait 2

# The gear in the top bar.
taptext "Settings" 06-settings
wait 3
shot 07-settings
dumpfile 07-settings
wait 2

# The themes are further down, so scroll.
swipe 540 1400 540 600 400 08-scrolled
wait 2
shot 09-settings-scrolled
dumpfile 09-settings-scrolled
wait 2

# The arrow back in the top bar of the settings.
taptext "Back" 10-back-from-the-settings
wait 3
shot 11-list-at-the-end
dumpfile 11-list-at-the-end
